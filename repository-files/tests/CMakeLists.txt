cmake_minimum_required(VERSION 3.22)
project(QTChatTests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Widgets Test)
enable_testing()
add_executable(qtchat-tests test-youtube.cpp ../src/youtube-chat.cpp ../src/youtube-chat.h ../src/twitch-irc.cpp ../src/twitch-irc.h log-stub.cpp)
target_include_directories(qtchat-tests PRIVATE ../src stubs)
target_link_libraries(qtchat-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
add_test(NAME youtube COMMAND qtchat-tests)
# Compile every plugin translation unit against real Qt/OBS headers when available.
find_path(OBS_INCLUDE_DIR obs-module.h PATH_SUFFIXES obs)
if(OBS_INCLUDE_DIR)
    file(GLOB PLUGIN_SOURCES ../src/*.cpp ../src/*.h)
    add_library(plugin-compile-check OBJECT ${PLUGIN_SOURCES})
    target_include_directories(plugin-compile-check PRIVATE ../src ${OBS_INCLUDE_DIR})
    target_link_libraries(plugin-compile-check PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Widgets)
endif()
add_executable(grpc-smoke grpc-smoke.cpp)
target_link_libraries(grpc-smoke PRIVATE Qt6::Core Qt6::Network)
target_include_directories(grpc-smoke PRIVATE ../src)
